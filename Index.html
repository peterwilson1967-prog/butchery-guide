    // SEARCH INPUT LISTENER

    document.getElementById('searchInput').addEventListener('input', (e) => {

      const query = e.target.value.toLowerCase().trim();

      if (!query) return;

      const foundKey = cutKeys.find(k => {

        const c = cutsData[k];

        return c.name.toLowerCase().includes(query) ||

               c.french.toLowerCase().includes(query) ||

               c.serviceDish.toLowerCase().includes(query) ||

               c.cookTech.toLowerCase().includes(query);

      });

      if (foundKey) {

        selectCut(foundKey);

      }

    });

    // QUIZ LOGIC

    function loadQuizQuestion() {

      const q = quizQuestions[currentQuizIndex];

      document.getElementById('quiz-progress').innerText = `Question ${currentQuizIndex + 1} of ${quizQuestions.length}`;

      document.getElementById('quiz-score').innerText = `Score: ${quizScore}`;

      document.getElementById('quiz-question').innerText = q.q;

      const feedbackBox = document.getElementById('quiz-feedback');

      feedbackBox.className = 'hidden';

      document.getElementById('quiz-next-btn').classList.add('hidden');

      const optionsBox = document.getElementById('quiz-options');

      optionsBox.innerHTML = q.options.map((opt, idx) => `

        <button onclick="handleQuizAnswer(${idx})" class="w-full text-left p-3.5 rounded-xl bg-slate-950 hover:bg-slate-800 border border-slate-800 text-xs text-slate-200 transition font-medium">

          <span class="inline-block w-5 font-mono text-orange-400 font-bold">${String.fromCharCode(65 + idx)}.</span> ${opt}

        </button>

      `).join('');

    }

    function handleQuizAnswer(selectedIdx) {

      const q = quizQuestions[currentQuizIndex];

      const feedbackBox = document.getElementById('quiz-feedback');

      const nextBtn = document.getElementById('quiz-next-btn');

      // Disable further clicks on option buttons

      document.querySelectorAll('#quiz-options button').forEach(b => b.disabled = true);

      if (selectedIdx === q.correct) {

        quizScore++;

        feedbackBox.className = 'p-3.5 rounded-xl text-xs font-medium border bg-emerald-950/70 text-emerald-300 border-emerald-700/80';

        feedbackBox.innerHTML = `<i class="fa-solid fa-circle-check mr-1.5"></i> <strong>Spot on Chef!</strong> ${q.explain}`;

      } else {

        feedbackBox.className = 'p-3.5 rounded-xl text-xs font-medium border bg-rose-950/70 text-rose-300 border-rose-700/80';

        feedbackBox.innerHTML = `<i class="fa-solid fa-circle-xmark mr-1.5"></i> <strong>Incorrect.</strong> ${q.explain}`;

      }

      document.getElementById('quiz-score').innerText = `Score: ${quizScore}`;

      nextBtn.classList.remove('hidden');

    }
